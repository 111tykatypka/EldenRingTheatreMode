cmake_minimum_required(VERSION 3.24)
project(EldenRingTheaterMode LANGUAGES CXX)
add_subdirectory(shared)

add_executable(EldenRingTheaterMode WIN32 src/monitor.cpp src/erplay.cpp src/replay_player.cpp)
target_compile_features(EldenRingTheaterMode PRIVATE cxx_std_20)
target_compile_definitions(EldenRingTheaterMode PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(EldenRingTheaterMode PRIVATE GameProfile Shell32 User32 Comctl32 Gdi32 Comdlg32)
if(MSVC)
  target_compile_options(EldenRingTheaterMode PRIVATE /W4 /permissive- /utf-8 /EHsc)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(erplay-tests src/erplay_tests.cpp src/erplay.cpp)
  target_compile_features(erplay-tests PRIVATE cxx_std_20)
  target_compile_definitions(erplay-tests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  if(MSVC)
    target_compile_options(erplay-tests PRIVATE /W4 /permissive- /utf-8 /EHsc)
  endif()
  add_test(NAME erplay-tests COMMAND erplay-tests)
  add_executable(replay-player-tests src/replay_player_tests.cpp src/replay_player.cpp src/erplay.cpp)
  target_compile_features(replay-player-tests PRIVATE cxx_std_20)
  if(MSVC)
    target_compile_options(replay-player-tests PRIVATE /W4 /permissive- /utf-8 /EHsc)
  endif()
  add_test(NAME replay-player-tests COMMAND replay-player-tests)
endif()
